function dshClaudeRootLifecycle(initialization) {
  if (Array.isArray(initialization?.capabilities) && initialization.capabilities.includes("msg_lifecycle_v1")) return true;
  // Verified on 2.1.325: initialize reports the version, but only system/init
  // advertises msg_lifecycle_v1, and it arrives AFTER the first started receipt.
  const match = /^(\d+)\.(\d+)\.(\d+)/.exec(initialization?.claude_code_version ?? "");
  return !!match && (+match[1] > 2 || +match[1] === 2 && (+match[2] > 1 || +match[2] === 1 && +match[3] >= 325));
}
class DshClaudeInputGate {
  constructor(id, required, text) {
    this.id = id; this.required = required; this.started = false;
    this.localCommand = /^\s*\//.test(text); this.hadRootOutput = false; this.cancelRequested = false;
  }
  inspect(message) {
    if (message?.type === "command_lifecycle" && message.command_uuid === this.id) {
      if (message.state === "queued") this.required = true;
      if (message.state === "started") { this.required = true; this.started = true; }
      if (!this.started && ["cancelled", "discarded", "refused", "completed"].includes(message.state)) return this.cancelRequested ? "cancelled" : this.localCommand ? "accept" : "rejected";
    }
    if (this.required && !this.started && !this.localCommand) {
      if (this.cancelRequested && message?.type === "result" && ["aborted_streaming", "aborted_tools"].includes(message.terminal_reason)) return "accept";
      return "wait";
    }
    if (message?.parent_tool_use_id == null && (message?.type === "assistant" && Array.isArray(message.message?.content) && message.message.content.length > 0 || message?.type === "stream_event" && message.event?.type === "message_start")) this.hadRootOutput = true;
    return "accept";
  }
  empty(terminal) { return terminal?.status === "succeeded" && !this.localCommand && !this.hadRootOutput; }
}
